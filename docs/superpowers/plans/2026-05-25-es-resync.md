> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在空间设置页加 "重新同步 ES 索引" 按钮，一键清空并重建本 space 的 4 类 ES 索引（project_draft / coze_resource / kb_entries / openynet_*），让运营人员自助修复 ES/MySQL 不同步。

**Tech Stack:** Go 1.25 (Hertz framework, gorm, sonic), TypeScript/React 18 (@coze-arch/coze-design, ahooks), Elasticsearch 8.x (bitnami/elasticsearch), Vitest (frontend), go test (backend).

```go
func (e *esStore) DeleteByQuery(ctx context.Context, index string, query map[string]any) (int64, error) {
	body, err := sonic.Marshal(map[string]any{"query": query})
	if err != nil {
		return 0, fmt.Errorf("marshal query: %w", err)
	}
	req := esapi.DeleteByQueryRequest{
		Index:    []string{index},
		Body:     bytes.NewReader(body),
		Refresh:  esapi.BoolPtr(true),
		// wait_for_completion=true (default) so we know deleted count synchronously
	}
	resp, err := req.Do(ctx, e.client)
	if err != nil {
		return 0, fmt.Errorf("delete_by_query %s: %w", index, err)
	}
	defer resp.Body.Close()
	if resp.IsError() {
		buf, _ := io.ReadAll(resp.Body)
		return 0, fmt.Errorf("delete_by_query %s: %s: %s", index, resp.Status(), string(buf))
	}
	var result struct {
		Deleted int64 `json:"deleted"`
	}
	if err := sonic.ConfigDefault.NewDecoder(resp.Body).Decode(&result); err != nil {
		return 0, fmt.Errorf("decode delete_by_query response: %w", err)
	}
	return result.Deleted, nil
}
```

```go
package service

import (
	"context"
	"fmt"

	"github.com/ynet-dev/ynet-studio/backend/api/model/data/space"
)

// ResyncSpace drops all ES docs for the given space + replays writes from MySQL
// for project_draft / coze_resource / kb_entries indices.
//
// Knowledge chunk indices (openynet_<kb_id>) are NOT handled here — they're
// the knowledge domain's responsibility (see knowledgeSvc.ResyncSpaceSlices).
func (s *searchSvc) ResyncSpace(ctx context.Context, spaceID int64) (*space.ResyncESCounts, error) {
	counts := &space.ResyncESCounts{}

	// (a) Fetch KB IDs first (needed for kb_entries delete_by_query terms filter)
	kbs, err := s.knowledgeRepo.ListBySpaceID(ctx, spaceID, 0)
	if err != nil {
		return nil, fmt.Errorf("list kbs by space: %w", err)
	}
	kbIDs := make([]int64, 0, len(kbs))
	for _, kb := range kbs {
		kbIDs = append(kbIDs, kb.ID)
	}

	// (b) delete_by_query on 3 list indices
	if _, err := s.searchStore.DeleteByQuery(ctx, "project_draft", map[string]any{
		"term": map[string]any{"space_id": spaceID},
	}); err != nil {
		return nil, fmt.Errorf("delete project_draft for space %d: %w", spaceID, err)
	}
	if _, err := s.searchStore.DeleteByQuery(ctx, "coze_resource", map[string]any{
		"term": map[string]any{"space_id": spaceID},
	}); err != nil {
		return nil, fmt.Errorf("delete coze_resource for space %d: %w", spaceID, err)
	}
	if len(kbIDs) > 0 {
		if _, err := s.searchStore.DeleteByQuery(ctx, "kb_entries", map[string]any{
			"terms": map[string]any{"kb_id": kbIDs},
		}); err != nil {
			return nil, fmt.Errorf("delete kb_entries for kbs %v: %w", kbIDs, err)
		}
	}

	// (c) Rewrite project_draft from agentRepo
	agents, err := s.agentRepo.ListBySpaceID(ctx, spaceID, 0)
	if err != nil {
		return nil, fmt.Errorf("list agents by space: %w", err)
	}
	for _, a := range agents {
		if err := s.indexAgent(ctx, a); err != nil {
			logs.CtxWarnf(ctx, "index agent %d failed: %v", a.AgentID, err)
			continue
		}
		counts.ProjectDraft++
	}

	// (d) Rewrite coze_resource from resourceRepo
	resources, err := s.resourceRepo.ListBySpaceID(ctx, spaceID, 0)
	if err != nil {
		return nil, fmt.Errorf("list resources by space: %w", err)
	}
	for _, r := range resources {
		if err := s.indexResource(ctx, r); err != nil {
			logs.CtxWarnf(ctx, "index resource %d failed: %v", r.ID, err)
			continue
		}
		counts.CozeResource++
	}

	// (e) Rewrite kb_entries from kbs (already fetched)
	for _, kb := range kbs {
		if err := s.indexKbEntry(ctx, kb); err != nil {
			logs.CtxWarnf(ctx, "index kb_entry %d failed: %v", kb.ID, err)
			continue
		}
		counts.KbEntries++
	}

	return counts, nil
}
```

注意：`indexAgent` / `indexResource` / `indexKbEntry` 这 3 个 helper 应该已经在 `handler_project.go` / `handler_resource.go` 等地方存在（被 event bus 复用）。**如果它们是私有 method（小写开头），在本任务的实现 commit 里一并把它们 promote 为同 package 内 exported helper**（首字母改大写 / 或重构成同 package 的 internal helper file），不留尾巴到下个 task。

```go
package service

import (
	"context"
	"fmt"
	"strconv"

	"github.com/bytedance/sonic"
	"github.com/cloudwego/eino/components/embedding/cache/redis-cache/eventbus"
	"github.com/ynet-dev/ynet-studio/backend/domain/knowledge/entity"
	"github.com/ynet-dev/ynet-studio/backend/domain/knowledge/internal/dal/model"
	"github.com/ynet-dev/ynet-studio/backend/domain/knowledge/internal/events"
	"github.com/ynet-dev/ynet-studio/backend/pkg/logs"
)

// ResyncSpaceSlices DELETEs each openynet_<kb_id> ES index for all KBs in the
// space, then sets each slice's status to Init and publishes IndexSliceEvent
// so the existing indexSlice consumer re-embeds + re-writes ES.
//
// Returns total count of slices queued for re-embedding.
func (k *knowledgeSVC) ResyncSpaceSlices(ctx context.Context, spaceID int64) (int, error) {
	kbs, err := k.knowledgeRepo.ListBySpaceID(ctx, spaceID, 0)
	if err != nil {
		return 0, fmt.Errorf("list kbs: %w", err)
	}

	total := 0
	for _, kb := range kbs {
		indexName := "openynet_" + strconv.FormatInt(kb.ID, 10)

		// Delete the entire ES index to drop stale vectors.
		// indexSlice consumer will re-create on first write.
		if err := k.searchStore.DeleteIndex(ctx, indexName); err != nil {
			logs.CtxWarnf(ctx, "delete ES index %s failed: %v (continuing)", indexName, err)
		}

		slices, err := k.sliceRepo.ListByKnowledgeID(ctx, kb.ID)
		if err != nil {
			logs.CtxWarnf(ctx, "list slices for kb %d failed: %v", kb.ID, err)
			continue
		}
		if len(slices) == 0 {
			continue
		}

		ids := make([]int64, 0, len(slices))
		for _, s := range slices {
			ids = append(ids, s.ID)
		}
		if err := k.sliceRepo.BatchSetStatus(ctx, ids, int32(model.SliceStatusInit), ""); err != nil {
			logs.CtxWarnf(ctx, "batch set status init for kb %d failed: %v", kb.ID, err)
			continue
		}

		for _, s := range slices {
			ev := events.NewIndexSliceEvent(&entity.Slice{
				Info:        knowledgeModel.Info{ID: s.ID},
				KnowledgeID: s.KnowledgeID,
				DocumentID:  s.DocumentID,
			}, nil) // document is fetched inside indexSlice consumer
			body, err := sonic.Marshal(ev)
			if err != nil {
				continue
			}
			if err := k.producer.Send(ctx, body, eventbus.WithShardingKey(strconv.FormatInt(s.DocumentID, 10))); err != nil {
				logs.CtxWarnf(ctx, "send IndexSliceEvent for slice %d failed: %v", s.ID, err)
				continue
			}
			total++
		}
	}
	return total, nil
}
```

```go
package space

import (
	"context"
	"fmt"

	spacemodel "github.com/ynet-dev/ynet-studio/backend/api/model/data/space"
	"github.com/ynet-dev/ynet-studio/backend/pkg/errorx"
	"github.com/ynet-dev/ynet-studio/backend/pkg/logs"
	"github.com/ynet-dev/ynet-studio/backend/types/errno"
)

func (s *SpaceApplicationService) ResyncES(ctx context.Context, req *spacemodel.ResyncESRequest) (*spacemodel.ResyncESResponse, error) {
	// 1. permission: caller must be space owner (or admin — TODO: hook into RBAC if available)
	currentUserID, ok := ctx.Value(userIDKey).(int64)
	if !ok || currentUserID == 0 {
		return nil, errorx.New(errno.ErrUnauthorizedCode, errorx.KV("msg", "not logged in"))
	}
	space, err := s.spaceRepo.GetByID(ctx, req.SpaceID)
	if err != nil {
		return nil, errorx.New(errno.ErrSpaceNotFoundCode, errorx.KV("msg", err.Error()))
	}
	if space == nil {
		return nil, errorx.New(errno.ErrSpaceNotFoundCode, errorx.KV("msg", fmt.Sprintf("space %d", req.SpaceID)))
	}
	if space.OwnerID != currentUserID {
		return nil, errorx.New(errno.ErrPermissionDeniedCode, errorx.KV("msg", "only space owner can resync ES"))
	}

	// 2. list-index rebuild (sync)
	counts, err := s.searchSvc.ResyncSpace(ctx, req.SpaceID)
	if err != nil {
		return nil, errorx.New(errno.ErrSpaceResyncESCode, errorx.KV("msg", err.Error()))
	}

	// 3. slice re-embedding (async via MQ)
	sliceJobs, err := s.knowledgeSvc.ResyncSpaceSlices(ctx, req.SpaceID)
	if err != nil {
		logs.CtxWarnf(ctx, "slice resync partial fail: %v (list indices already rebuilt)", err)
	}
	counts.SliceReindexJobs = sliceJobs

	return &spacemodel.ResyncESResponse{
		Code:   0,
		Msg:    "",
		Counts: counts,
	}, nil
}
```

```tsx
import { useState } from 'react';
import { Toast, Button, Modal, Card } from '@coze-arch/coze-design';
import { SpaceApi } from '@coze-arch/bot-api';
import { I18n } from '@coze-arch/i18n';
import styles from './DataMaintenanceSection.module.less';

interface Props {

}

export const DataMaintenanceSection: React.FC<Props> = ({ spaceId }) => {
  const [confirmOpen, setConfirmOpen] = useState(false);
  const [loading, setLoading] = useState(false);

  const handleResync = async () => {
    setLoading(true);
    try {
      const resp = await SpaceApi.ResyncES({ space_id: spaceId });
      if (resp.code !== 0) throw new Error(resp.msg || 'unknown');
      const { project_draft, coze_resource, kb_entries, slice_reindex_jobs } = resp.counts;
      Toast.success(

          agents: project_draft,
          resources: coze_resource,
          kbs: kb_entries,
          slices: slice_reindex_jobs,
        })
      );
    } catch (e: any) {
      Toast.error(I18n.t('space_resync_failed', { msg: e?.message || 'unknown error' }));
    } finally {
      setLoading(false);
      setConfirmOpen(false);
    }
  };

  return (
    <section className={styles.section}>
      <h3 className={styles.title}>{I18n.t('space_data_maintenance')}</h3>
      <Card className={styles.card}>
        <div className={styles.cardTitle}>{I18n.t('space_resync_es_title')}</div>
        <div className={styles.cardDesc}>{I18n.t('space_resync_es_desc')}</div>
        <Button
          danger
          loading={loading}
          onClick={() => setConfirmOpen(true)}
          data-testid="space-resync-es-button"
        >
          {I18n.t('space_resync_es_button')}
        </Button>
      </Card>
      <Modal
        visible={confirmOpen}
        title={I18n.t('space_resync_es_confirm_title')}
        onOk={handleResync}
        okText={I18n.t('confirm')}
        cancelText={I18n.t('cancel')}
        confirmLoading={loading}
        onCancel={() => setConfirmOpen(false)}
      >
        <p>{I18n.t('space_resync_es_confirm_desc')}</p>
      </Modal>
    </section>
  );
};
```
